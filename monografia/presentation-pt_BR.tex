\begin{frame}{Motivação \& pergunta}
  \begin{itemize}
    \item Forex pode reagir a notícias e choques geopolíticos em segundos; estratégias só de preço ignoram isso quando ocorre.
    \item \textbf{Pergunta}: sentimento + eventos superam um \textit{baseline} só de preço? Testado, não assumido.
  \end{itemize}
\end{frame}

\begin{frame}{Protocolo de avaliação}
  \begin{itemize}
    \item \textit{Walk-forward} + $k$-fold purgado com \textit{embargo} (López de Prado).
    \item Uma janela de teste pré-registrada e intocada por reivindicação.
    \item vs.\ trabalho mais próximo (Zhang 2025): protocolo de validação diferente, não diretamente comparável.
  \end{itemize}
\end{frame}

\begin{frame}{Por quê -- três causas auditadas}
  \begin{itemize}
    \item Limiares calibrados em um mês não sobreviveram a uma mudança de patamar.
    \item O meta-modelo não carregou informação em todo o ano.
    \item O sinal de notícias existe, mas em horizonte de 4h; a estratégia segura por dias.
  \end{itemize}
\end{frame}

\begin{frame}{Dois desdobramentos executados este ano}
  \begin{itemize}
    \item \textbf{Retreinamento adaptativo}: modelo congelado, ano completo -- $-97.8\%$, 26\% de acerto. 4/5 políticas seguem pendentes.
    \item \textbf{Catálogo de candlestick} (22 padrões): um positivo isolado, indistinguível do acaso (teste de permutação, 98.7\%). Catálogo inteiro: $-99.5\%$ / $-32.8\%$.
  \end{itemize}
\end{frame}

\begin{frame}{Conclusão \& trabalhos futuros}
  \begin{itemize}
