\begin{frame}{Motivation \& question}
  \begin{itemize}
    \item Forex can react to news and geopolitical shocks in seconds; price-only strategies ignore that when it happens.
    \item \textbf{Question}: does sentiment + events beat a price-only baseline? Tested, not assumed.
  \end{itemize}
\end{frame}

      \item \alert{No arm beats the baseline} ($p=0.149$).

\begin{frame}{Why -- three audited causes}
  \begin{itemize}
    \item Thresholds calibrated on one month didn't survive a level shift.
    \item The meta-learner carried no information all year.
    \item The news signal exists, but on a 4h horizon; the strategy holds for days.
  \end{itemize}
\end{frame}

\begin{frame}{Two follow-ups run this year}
  \begin{itemize}
    \item \textbf{Adaptive retraining}: frozen model, full year -- $-97.8\%$, 26\% win rate. 4/5 policies still outstanding.
    \item \textbf{Candlestick catalog} (22 patterns): one positive alone, indistinguishable from chance (permutation test, 98.7\%). Whole catalog: $-99.5\%$ / $-32.8\%$.
  \end{itemize}
\end{frame}

\begin{frame}{Does adding GDELT news help the one positive pattern?}

\begin{frame}{Conclusion \& future work}
  \begin{itemize}
    \item Pipeline and execution layer work end to end, proven not assumed.
    \item No configuration tested this year beat the price-only baseline.
    \item Registered, not yet run: confluence chain, intermarket features, price-action/Ichimoku contest, 4 retraining policies.
  \end{itemize}
\end{frame}
